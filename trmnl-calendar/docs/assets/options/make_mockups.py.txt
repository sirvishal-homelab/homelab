def tkey(t):
    if not t: return -1
    h, m = t[:-1].split(":") if ":" in t else (t[:-1], "0")
    h = int(h) % 12 + (12 if t[-1] == "p" else 0); return h * 60 + int(m)
def day_events(off, cat=None):
    out = []
    for e in EV:
        t, o, tm, c, ppl, own, span = e
        if o <= off < o + span and (cat is None or c == cat):
            out.append(dict(title=t, time=tm if off == o else None, cat=c, people=ppl, owner=own, day=off - o + 1, span=span, start=o))
    return sorted(out, key=lambda x: (tkey(x["time"]), x["title"]))

def rng(off, span):
    a = START + dt.timedelta(days=off); b = START + dt.timedelta(days=off + span - 1)
    return f"{a.strftime('%b')} {a.day}-{b.day}" if a.month == b.month else f"{a.strftime('%b')} {a.day}-{b.strftime('%b')} {b.day}"
def segments(week):
    segs = []
    for t, o, tm, c, ppl, own, span in EV:
        if span <= 1: continue
        a, b = max(o, week * 7), min(o + span - 1, week * 7 + 6)
        if a <= b: segs.append(dict(title=t, a=a - week * 7, b=b - week * 7, cl=o < week * 7, cr=o + span - 1 > week * 7 + 6, people=ppl, owner=own, first=(o >= week * 7), start=o, span=span))
    segs.sort(key=lambda z: (z["a"], -(z["b"] - z["a"]))); ends = []
    for z in segs:
        for li, en in enumerate(ends):
            if z["a"] > en: ends[li] = z["b"]; z["lvl"] = li; break
        else: ends.append(z["b"]); z["lvl"] = len(ends) - 1
    return segs, len(ends)
class C:
    def __init__(s):
        s.im = Image.new("L", (W, H), 255); s.d = ImageDraw.Draw(s.im); s.d.fontmode = "1"
    def text(s, xy, t, f, fill=0, anchor="la"): s.d.text(xy, t, font=f, fill=fill, anchor=anchor)
    def tw(s, t, f): return s.d.textlength(t, font=f)
    def ell(s, t, f, mw):
        if s.tw(t, f) <= mw: return t
        while t and s.tw(t + "…", f) > mw: t = t[:-1]
        return t.rstrip() + "…"
    def wrap(s, t, f, mw, ml):
        words, lines, cur = t.split(), [], ""
        for w in words:
            x = (cur + " " + w).strip()
            if s.tw(x, f) <= mw: cur = x
            else:
                if cur: lines.append(cur)
                cur = w
        if cur: lines.append(cur)
        if len(lines) > ml:
            rest = " ".join(lines[ml - 1:]); lines = lines[:ml - 1] + [s.ell(rest, f, mw)]
        return [s.ell(l, f, mw) for l in lines]
    def dotted_h(s, x1, x2, y, step=3):
        for x in range(int(x1), int(x2), step * 2): s.d.line((x, y, x + step - 1, y), fill=0)
    def dotted_v(s, x, y1, y2, step=3):
        for y in range(int(y1), int(y2), step * 2): s.d.line((x, y, x, y + step - 1), fill=0)
    def glyph(s, x, y, cat, z=8):       # x,y top-left; z size
        d = s.d
        if cat == "Events": d.rectangle((x, y, x + z - 2, y + z - 2), fill=0)
        elif cat == "Work": d.polygon([(x + z / 2 - .5, y), (x + z - 1, y + z - 1), (x, y + z - 1)], fill=0)
        elif cat == "Lunch": d.ellipse((x, y, x + z - 1, y + z - 1), fill=0)
        else: d.polygon([(x + z / 2 - .5, y), (x + z - 1, y + z / 2 - .5), (x + z / 2 - .5, y + z - 1), (x, y + z / 2 - .5)], fill=0)
    def badges(s, x, y, e, right=False):   # boxed initials; owner filled grey. returns width used
        lst = ([e["owner"]] if e["owner"] else []) + [p for p in e["people"] if p != e["owner"]]
        w = 12 * len(lst) + (len(lst) - 1) * 1 if lst else 0
        if right: x = x - w
        for i, p in enumerate(lst):
            bx = x + i * 13
            s.d.rectangle((bx, y, bx + 11, y + 11), outline=0, fill=170 if p == e["owner"] else 255)
            s.text((bx + 6, y + 10), p, F(10, True), anchor="ms")
        return w
    def save(s, name):
        q = s.im.point(lambda v: min(range(4), key=lambda i: abs(LV[i] - v))).convert("P")
        pal = []
        for v in LV: pal += [v, v, v]
        q.putpalette(pal + [0] * (768 - len(pal))); q.save(name, bits=2, optimize=True)
        s.im.save(name.replace(".png", "_preview.png"))
def label(e): return (e["time"] + " " if e["time"] else "") + e["title"]
def title(c, t, right=None):
    c.text((14, 8), t, F(22, True))
    if right: c.text((W - 14, 14), right, F(11), anchor="ra")
def footer(c, left=None):
    if left: c.text((14, H - 6), left, F(11), anchor="ls")
    c.text((W - 14, H - 6), "Updated 9:00 AM", F(11), anchor="rs")
def legend(c, x, y):
    for n in CATS:
        c.glyph(x, y + 1, n); c.text((x + 12, y + 9), n, F(11), anchor="ls"); x += 14 + c.tw(n, F(11)) + 14
def dayname(off): return (START + dt.timedelta(days=off)).strftime("%a")

# ---------- two-week grids
def twoweek(mode, trips=False):
    c = C(); m = 12; cw = (W - 2 * m) // 7
    sub = {"lanes": "Two weeks, lanes in each day (top to bottom: categories)", "flow": "Two weeks, events flow with category symbols" + (", trips as bars" if trips else "")}[mode]
    title(c, "Oct 25 – Nov 7, 2026", sub)
    hy = 44
    for i in range(7): c.text((m + i * cw + cw / 2, hy + 12), dayname(i), F(12, True), anchor="ms")
    c.d.line((m, hy + 17, W - m, hy + 17), fill=0, width=2)
    top = hy + 19; bot = H - 22; rh = (bot - top) // 2
    for wk in range(2):
        y = top + wk * rh
        if wk: c.d.line((m, y, W - m, y), fill=0)
        segs, nl = segments(wk) if (mode == "flow") else ([], 0)
        for i in range(7):
            off = wk * 7 + i; x = m + i * cw
            if i: c.dotted_v(x, y + 2, y + rh - 2)
            numbox = (x + 3, y + 2, x + 25, y + 19)
            if off == 0: c.d.rectangle(numbox, fill=170)
            d = START + dt.timedelta(days=off)
            c.text((x + 5, y + 16), (d.strftime("%b %-d") if d.day == 1 else str(d.day)), F(13, True), anchor="ls")
            if off in HOL:
                bx0 = x + (50 if d.day == 1 else 38); c.d.rectangle((bx0, y + 4, x + cw - 3, y + 17), fill=170); c.text((bx0 + 3, y + 15), HOL[off][:11], F(10, True), anchor="ls")
            body_top = y + 21 + (nl * 16 + 2 if nl else 0); body_h = rh - 23 - (nl * 16 + 2 if nl else 0)
            if mode == "lanes":
                lh = body_h // 4
                for li, cat in enumerate(CATS):
                    ty = body_top + li * lh
                    if li: c.dotted_h(x + 2, x + cw - 2, ty)
                    evs = [e for e in day_events(off, cat) if e["span"] == 1]
                    if not evs: continue
                    e = evs[0]; c.glyph(x + 4, ty + 4, cat, 7); extra = len(evs) - 1
                    lines = c.wrap(label(e), F(11), cw - 20, 2 if extra == 0 else 1)
                    for k, l in enumerate(lines): c.text((x + 13, ty + 11 + k * 12), l, F(11), anchor="ls")
                    if extra: c.text((x + cw - 4, ty + lh - 3), f"+{extra}", F(11, True), anchor="rs")
            else:
                evs = sorted([e for e in day_events(off) if e["span"] == 1], key=lambda e: (tkey(e["time"]), CATS.index(e["cat"])))
                yy = body_top; maxlines = body_h // 13; used = 0
                for j, e in enumerate(evs):
                    lines = c.wrap(label(e), F(11), cw - 20, 2)
                    if used + len(lines) > maxlines - (1 if j < len(evs) - 1 else 0):
                        c.text((x + 5, yy + used * 13 + 10), f"+{len(evs) - j} more", F(11, True), anchor="ls"); break
                    c.glyph(x + 4, yy + used * 13 + 3, e["cat"], 7)
                    for k, l in enumerate(lines): c.text((x + 13, yy + used * 13 + 10 + k * 13), l, F(11), anchor="ls")
                    used += len(lines)
        for z in segs:
            x1 = m + z["a"] * cw + 3; x2 = m + (z["b"] + 1) * cw - 3; yb = y + 21 + z["lvl"] * 16
            c.d.rectangle((x1, yb, x2, yb + 14), fill=170, outline=0)
            t = ("◂ " if z["cl"] else "") + z["title"] + ((" · " + rng(z["start"], z["span"])) if z["first"] else "") + (" ▸" if z["cr"] else "")
            c.text((x1 + 4, yb + 12), c.ell(t, F(11, True), x2 - x1 - 8), F(11, True), anchor="ls")
    c.d.line((m, bot, W - m, bot), fill=0)
    if mode == "flow": legend(c, 14, H - 18); c.text((W - 14, H - 6), "Updated 9:00 AM", F(11), anchor="rs")
    else: footer(c, "Lanes in each day, top to bottom: ■ Events  ▲ Work  ● Lunch  ◆ Dinner")
    return c
